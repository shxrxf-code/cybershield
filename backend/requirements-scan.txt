# CyberShield backend - signature-based scanning dependencies
#
# Install with:  pip install -r requirements.txt -r requirements-scan.txt
#
# The application degrades safely when these are absent: the YARA engine reports
# itself as unavailable, lowers scan coverage, and reduces verdict confidence
# rather than silently reporting a clean result.

# YARA rule engine for defensive pattern matching.
yara-python>=4.3,<5.0

# libmagic bindings for supplementary MIME detection. CyberShield's built-in
# signature table remains authoritative, so libmagic is an optional cross-check.
python-magic>=0.4.27,<0.5
